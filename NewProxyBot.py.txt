# ── Data Storage ──
DATA_FILE = "proxy_data.json"

def load_data():
    """Load data from JSON file"""
    try:
        with open(DATA_FILE, 'r') as f:
            return json.load(f)
    except FileNotFoundError:
        return {
            "active_keys": [],
            "used_keys": [],
            "resellers": [],
            "banned_keys": [],
            "key_counter": 0
        }

def save_data(data):
    """Save data to JSON file"""
    with open(DATA_FILE, 'w') as f:
        json.dump(data, f, default=str, indent=2)

# Load data
db = load_data()

# ── Helper Functions ──
def gen_key():
    db["key_counter"] += 1
    return "".join(random.choices(string.ascii_uppercase + string.digits, k=KEY_LENGTH))

def clean_expired_keys():
    """Remove expired keys from active_keys."""
    now = datetime.now()
    db["active_keys"] = [k for k in db["active_keys"] if datetime.fromisoformat(k["expiry"]) > now]
    save_data(db)

def is_key_banned(key):
    return key in db["banned_keys"]

def is_reseller(user_id):
    return any(r["user_id"] == user_id for r in db["resellers"])

def get_reseller(user_id):
    for r in db["resellers"]:
        if r["user_id"] == user_id:
            return r
    return None

def format_key_info(key_data):
    """Format key info for display"""
    exp = datetime.fromisoformat(key_data["expiry"]).strftime("%Y-%m-%d %H:%M")
    return f"`{key_data['key']}` – expires {exp}"

# ── Handlers ──
async def start(update: Update, ctx: ContextTypes.DEFAULT_TYPE):
    clean_expired_keys()
    active_users = len(db["active_keys"])
    welcome_text = WELCOME.format(active_users=active_users)
    await update.message.reply_text(welcome_text, reply_markup=main_keyboard(active_users), parse_mode="Markdown")

async def admin_cmd(update: Update, ctx: ContextTypes.DEFAULT_TYPE):
    if update.effective_user.id != OWNER_ID:
        await update.message.reply_text("⛔ Access denied.", parse_mode="Markdown")
        return
    await update.message.reply_text(
        "🔐 *Admin Panel - MESTRAX Proxy*\n\nChoose action:",
        reply_markup=admin_keyboard(),
        parse_mode="Markdown",
    )

async def on_button(update: Update, ctx: ContextTypes.DEFAULT_TYPE):
    query = update.callback_query
    await query.answer()
    uid = query.from_user.id
    data = query.data

    # ── Global navigation ──
    if data == "main_menu":
        active_users = len(db["active_keys"])
        menu_text = MAIN_MENU.format(active_users=active_users)
        await query.edit_message_text(menu_text, reply_markup=main_keyboard(active_users), parse_mode="Markdown")
        ctx.user_data["state"] = "menu"

    elif data == "activate":
        await query.edit_message_text(ASK_KEY, reply_markup=back_keyboard(), parse_mode="Markdown")
        ctx.user_data["state"] = "waiting_key"

    elif data == "check_ip":
        await query.edit_message_text(
            CHECK_IP,
            reply_markup=InlineKeyboardMarkup([
                [InlineKeyboardButton("🔗 Show My IP", url="https://api.ipify.org?format=json")],
                [InlineKeyboardButton("🔙 Back", callback_data="main_menu")],
            ]),
            parse_mode="Markdown"
        )

    elif data == "stats":
        await query.edit_message_text(STATS_MSG, reply_markup=back_keyboard(), parse_mode="Markdown")

    elif data == "help":
        await query.edit_message_text(HELP_MSG, reply_markup=back_keyboard(), parse_mode="Markdown")

    elif data == "ask_pass":
        await query.edit_message_text(ASK_PASSWORD, reply_markup=back_keyboard(), parse_mode="Markdown")
        ctx.user_data["state"] = "waiting_pass"

    elif data == "admin_panel":
        await query.edit_message_text(
            "🔐 *Admin Panel - MESTRAX Proxy*\n\nChoose action:",
            reply_markup=admin_keyboard(),
            parse_mode="Markdown",
        )

    # ── Admin actions ──
    elif data == "gen_keys":
        if uid != OWNER_ID:
            await query.answer("⛔ Not authorized!", show_alert=True)
            return
        await query.edit_message_text(ASK_KEY_COUNT, reply_markup=cancel_keyboard(), parse_mode="Markdown")
        ctx.user_data["state"] = "waiting_key_count"

    elif data == "manage_resellers":
        if uid != OWNER_ID:
            await query.answer("⛔ Not authorized!", show_alert=True)
            return
        await query.edit_message_text(
            "👥 *Manage Resellers*\n\nChoose an option:",
            reply_markup=manage_resellers_keyboard(),
            parse_mode="Markdown"
        )

    elif data == "add_reseller":
        if uid != OWNER_ID:
            await query.answer("⛔ Not authorized!", show_alert=True)
            return
        await query.edit_message_text(ASK_RESELLER_ID, reply_markup=cancel_keyboard(), parse_mode="Markdown")
        ctx.user_data["state"] = "waiting_reseller_id"

    elif data == "list_resellers":
        if uid != OWNER_ID:
            await query.answer("⛔ Not authorized!", show_alert=True)
            return
        if not db["resellers"]:
            await query.edit_message_text(
                "📋 *No resellers registered.*",
                reply_markup=manage_resellers_keyboard(),
                parse_mode="Markdown"
            )
        else:
            await query.edit_message_text(
                "📋 *List of Resellers:*\n\nSelect a reseller for details:",
                reply_markup=reseller_list_keyboard(db["resellers"]),
                parse_mode="Markdown"
            )

    elif data.startswith("reseller_info_"):
        if uid != OWNER_ID:
            await query.answer("⛔ Not authorized!", show_alert=True)
            return
        reseller_id = int(data.split("_")[2])
        reseller = get_reseller(reseller_id)
        if not reseller:
            await query.edit_message_text(
                "❌ Reseller not found.",
                reply_markup=manage_resellers_keyboard(),
                parse_mode="Markdown"
            )
            return
        # Get stats
        keys_generated = reseller.get("keys_generated", 0)
        keys_used = reseller.get("keys_used", 0)
        await query.edit_message_text(
            f"👤 *Reseller Details*\n\n"
            f"📝 Name: {reseller['name']}\n"
            f"🆔 ID: {reseller['user_id']}\n"
            f"📊 Keys Generated: {keys_generated}\n"
            f"📦 Keys Used: {keys_used}\n"
            f"📈 Total: {keys_generated + keys_used}",
            reply_markup=manage_resellers_keyboard(),
            parse_mode="Markdown"
        )

    elif data == "remove_reseller":
        if uid != OWNER_ID:
            await query.answer("⛔ Not authorized!", show_alert=True)
            return
        if not db["resellers"]:
            await query.edit_message_text(
                "📋 *No resellers to remove.*",
                reply_markup=manage_resellers_keyboard(),
                parse_mode="Markdown"
            )
            return
        keyboard = []
        for r in db["resellers"]:
            keyboard.append([InlineKeyboardButton(
                f"❌ {r['name']} (ID: {r['user_id']})",
                callback_data=f"remove_reseller_{r['user_id']}"
            )])
        keyboard.append([InlineKeyboardButton("🔙 Back", callback_data="manage_resellers")])
        await query.edit_message_text(
            "❌ *Select a reseller to remove:*",
            reply_markup=InlineKeyboardMarkup(keyboard),
            parse_mode="Markdown"
        )

    elif data.startswith("remove_reseller_"):
        if uid != OWNER_ID:
            await query.answer("⛔ Not authorized!", show_alert=True)
            return
        reseller_id = int(data.split("_")[2])
        db["resellers"] = [r for r in db["resellers"] if r["user_id"] != reseller_id]
        save_data(db)
        await query.edit_message_text(
            "✅ *Reseller removed successfully!*",
            reply_markup=manage_resellers_keyboard(),
            parse_mode="Markdown"
        )

    # ── Admin View Keys ──
    elif data == "view_keys":
        if uid != OWNER_ID:
            await query.answer("⛔ Not authorized!", show_alert=True)
            return
        clean_expired_keys()
        if not db["active_keys"]:
            msg = "📦 *No active keys.*\n\nGenerate new keys first."
        else:
            lines = []
            for idx, k in enumerate(db["active_keys"][:50], 1):  # Limit to 50
                lines.append(f"{idx}. {format_key_info(k)}")
            msg = f"📦 *Active Keys ({len(db['active_keys'])} total)*\n\n" + "\n".join(lines)
            if len(db["active_keys"]) > 50:
                msg += f"\n\n... and {len(db['active_keys']) - 50} more"
        await query.edit_message_text(msg, reply_markup=admin_keyboard(), parse_mode="Markdown")

    # ── Admin Ban/Unban ──
    elif data == "ban_key":
        if uid != OWNER_ID:
            await query.answer("⛔ Not authorized!", show_alert=True)
            return
        await query.edit_message_text(
            "🚫 *Enter the key to ban:*\n\nType the key in the chat:",
            reply_markup=cancel_keyboard(),
            parse_mode="Markdown"
        )
        ctx.user_data["state"] = "waiting_ban_key"

    elif data == "unban_key":
        if uid != OWNER_ID:
            await query.answer("⛔ Not authorized!", show_alert=True)
            return
        if not db["banned_keys"]:
            await query.edit_message_text(
                "📋 *No banned keys.*",
                reply_markup=admin_keyboard(),
                parse_mode="Markdown"
            )
            return
        keyboard = []
        for key in db["banned_keys"][:20]:
            keyboard.append([InlineKeyboardButton(
                f"🔓 {key}",
                callback_data=f"unban_{key}"
            )])
        keyboard.append([InlineKeyboardButton("🔙 Back", callback_data="admin_panel")])
        await query.edit_message_text(
            "🔓 *Select a key to unban:*",
            reply_markup=InlineKeyboardMarkup(keyboard),
            parse_mode="Markdown"
        )

    elif data.startswith("unban_"):
        if uid != OWNER_ID:
            await query.answer("⛔ Not authorized!", show_alert=True)
            return
        key = data.split("_", 1)[1]
        if key in db["banned_keys"]:
            db["banned_keys"].remove(key)
            save_data(db)
            await query.edit_message_text(
                f"✅ *Key unbanned successfully!*\n\n`{key}`",
                reply_markup=admin_keyboard(),
                parse_mode="Markdown"
            )
        else:
            await query.edit_message_text(
                "❌ Key not found in banned list.",
                reply_markup=admin_keyboard(),
                parse_mode="Markdown"
            )

    # ── Admin Delete Keys ──
    elif data == "del_keys":
        if uid != OWNER_ID:
            await query.answer("⛔ Not authorized!", show_alert=True)
            return
        db["active_keys"].clear()
        save_data(db)
        await query.edit_message_text(
            "🗑 *All keys deleted.*",
            reply_markup=admin_keyboard(),
            parse_mode="Markdown",
        )

    # ── Admin Reseller Stats ──
    elif data == "reseller_stats":
        if uid != OWNER_ID:
            await query.answer("⛔ Not authorized!", show_alert=True)
            return
        if not db["resellers"]:
            await query.edit_message_text(
                "📊 *No resellers registered.*",
                reply_markup=admin_keyboard(),
                parse_mode="Markdown"
            )
            return
        msg = "📊 *Reseller Statistics*\n\n"
        for r in db["resellers"]:
            keys_gen = r.get("keys_generated", 0)
            keys_used = r.get("keys_used", 0)
            msg += f"👤 {r['name']} (ID: {r['user_id']})\n"
            msg += f"   📊 Generated: {keys_gen}\n"
            msg += f"   📦 Used: {keys_used}\n\n"
        await query.edit_message_text(msg, reply_markup=admin_keyboard(), parse_mode="Markdown")

    # ── Reseller actions ──
    elif data.startswith("reseller_"):
        if not is_reseller(uid):
            await query.answer("⛔ Access denied! You are not a reseller.", show_alert=True)
            return
        reseller = get_reseller(uid)
        
        if data == "reseller_gen_keys":
            await query.edit_message_text(ASK_KEY_COUNT, reply_markup=cancel_keyboard(), parse_mode="Markdown")
            ctx.user_data["state"] = "reseller_waiting_key_count"
            ctx.user_data["reseller_id"] = uid

        elif data == "reseller_view_keys":
            clean_expired_keys()
            reseller_keys = [k for k in db["active_keys"] if k.get("generated_by") == uid]
            if not reseller_keys:
                msg = "📦 *No active keys.*\n\nGenerate new keys first."
            else:
                lines = []
                for idx, k in enumerate(reseller_keys[:50], 1):
                    lines.append(f"{idx}. {format_key_info(k)}")
                msg = f"📦 *Your Active Keys ({len(reseller_keys)} total)*\n\n" + "\n".join(lines)
                if len(reseller_keys) > 50:
                    msg += f"\n\n... and {len(reseller_keys) - 50} more"
            await query.edit_message_text(msg, reply_markup=reseller_keyboard(), parse_mode="Markdown")

        elif data == "reseller_stats":
            msg = f"📊 *Reseller Stats*\n\n"
            msg += f"👤 Name: {reseller['name']}\n"
            msg += f"🆔 ID: {reseller['user_id']}\n"
            msg += f"📊 Keys Generated: {reseller.get('keys_generated', 0)}\n"
            msg += f"📦 Keys Used: {reseller.get('keys_used', 0)}\n"
            msg += f"📈 Total: {reseller.get('keys_generated', 0) + reseller.get('keys_used', 0)}"
            await query.edit_message_text(msg, reply_markup=reseller_keyboard(), parse_mode="Markdown")

    # ── Duration selection ──
    elif data.startswith("dur_"):
        if uid != OWNER_ID and not is_reseller(uid):
            await query.answer("⛔ Not authorized!", show_alert=True)
            return
        days = int(data.split("_")[1])
        count = ctx.user_data.get("key_count", 1)
        generated_keys = []
        
        for _ in range(count):
            key = gen_key()
            expiry = datetime.now() + timedelta(days=days)
            key_data = {
                "key": key,
                "expiry": expiry.isoformat(),
                "duration": f"{days} days",
                "generated_by": uid
            }
            db["active_keys"].append(key_data)
            generated_keys.append(key)
        
        # Update reseller stats if applicable
        if is_reseller(uid):
            reseller = get_reseller(uid)
            reseller["keys_generated"] = reseller.get("keys_generated", 0) + count
            save_data(db)
        
        save_data(db)
        
        keys_display = "\n".join([f"`{k}`" for k in generated_keys[:10]])
        if len(generated_keys) > 10:
            keys_display += f"\n... and {len(generated_keys) - 10} more"
        
        await query.edit_message_text(
            f"🔑 *{count} Key(s) Generated!*\n\n"
            f"{keys_display}\n\n"
            f"⏳ Expires: {(datetime.now() + timedelta(days=days)).strftime('%Y-%m-%d %H:%M:%S')}\n"
            f"📅 Duration: {days} days\n\n"
            f"📊 Active: {len(db['active_keys'])} | Used: {len(db['used_keys'])}",
            reply_markup=gen_keyboard() if uid == OWNER_ID else reseller_gen_keyboard(),
            parse_mode="Markdown",
        )
        ctx.user_data["state"] = "owner" if uid == OWNER_ID else "reseller"

    # ── Cancel ──
    elif data == "cancel":
        await query.edit_message_text(
            "❌ Operation cancelled.",
            reply_markup=main_keyboard(len(db["active_keys"])),
            parse_mode="Markdown"
        )
        ctx.user_data["state"] = "menu"


async def on_message(update: Update, ctx: ContextTypes.DEFAULT_TYPE):
    state = ctx.user_data.get("state", "menu")
    uid = update.effective_user.id
    text = update.message.text.strip()

    # ── Password check ──
    if state == "waiting_pass":
        if text == OWNER_PASSWORD and uid == OWNER_ID:
            await update.message.reply_text(
                "✅ *Access Granted*\n\n🔐 Admin Panel:",
                reply_markup=admin_keyboard(),
                parse_mode="Markdown",
            )
            ctx.user_data["state"] = "owner"
        else:
            await update.message.reply_text(
                "❌ *Wrong password.*",
                reply_markup=main_keyboard(len(db["active_keys"])),
                parse_mode="Markdown",
            )
            ctx.user_data["state"] = "menu"
        return

    # ── Waiting for key count ──
    if state == "waiting_key_count":
        if uid != OWNER_ID:
            await update.message.reply_text("⛔ Not authorized!")
            return
        try:
            count = int(text)
            if count < 1 or count > 1000:
                await update.message.reply_text(
                    "❌ *Invalid number.* Please enter a number between 1 and 1000.",
                    parse_mode="Markdown"
                )
                return
            ctx.user_data["key_count"] = count
            await update.message.reply_text(
                ASK_DURATION,
                reply_markup=duration_keyboard(),
                parse_mode="Markdown"
            )
            ctx.user_data["state"] = "waiting_duration"
        except ValueError:
            await update.message.reply_text(
                "❌ *Invalid input.* Please enter a number.",
                parse_mode="Markdown"
            )
        return
