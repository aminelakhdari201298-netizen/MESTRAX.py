def init_db():
    conn = sqlite3.connect('store.db', check_same_thread=False)
    c = conn.cursor()
    
    # جدول المستخدمين (بدون كلمة مرور)
    c.execute('''CREATE TABLE IF NOT EXISTS users 
                 (telegram_id INTEGER PRIMARY KEY, username TEXT, balance REAL DEFAULT 0.0, 
                  join_date TEXT, is_admin INTEGER DEFAULT 0)''')
    
    # جدول المنتجات
    c.execute('''CREATE TABLE IF NOT EXISTS products 
                 (id INTEGER PRIMARY KEY AUTOINCREMENT, name TEXT, price REAL, 
                  category TEXT, stock INTEGER DEFAULT 0)''')
    
    # جدول المفاتيح
    c.execute('''CREATE TABLE IF NOT EXISTS keys 
                 (id INTEGER PRIMARY KEY AUTOINCREMENT, product_id INTEGER, 
                  key_data TEXT, status TEXT DEFAULT 'available', 
                  purchase_date TEXT, buyer_id INTEGER)''')
    
    # جدول المشتريات
    c.execute('''CREATE TABLE IF NOT EXISTS purchases 
                 (id INTEGER PRIMARY KEY AUTOINCREMENT, user_id INTEGER, 
                  product_id INTEGER, key_id INTEGER, purchase_date TEXT, 
                  amount REAL)''')
    
    # إضافة الأدمن إذا لم يكن موجوداً
    c.execute("INSERT OR IGNORE INTO users (telegram_id, username, is_admin) VALUES (?, ?, ?)", 
              (ADMIN_ID, 'Admin', 1))
    
    conn.commit()
    conn.close()

init_db()

def execute_query(query, params=()):
    conn = sqlite3.connect('store.db', check_same_thread=False)
    c = conn.cursor()
    c.execute(query, params)
    conn.commit()
    conn.close()

def fetch_query(query, params=()):
    conn = sqlite3.connect('store.db', check_same_thread=False)
    c = conn.cursor()
    c.execute(query, params)
    result = c.fetchall()
    conn.close()
    return result

def get_user(telegram_id):
    result = fetch_query("SELECT * FROM users WHERE telegram_id=?", (telegram_id,))
    return result[0] if result else None

def is_admin(telegram_id):
    user = get_user(telegram_id)
    return user and user[4] == 1

@bot.message_handler(func=lambda message: message.text == '📦 مشترياتي')
def my_purchases(message):
    user = get_user(message.chat.id)
    if not user:
        bot.send_message(message.chat.id, "⚠️ يرجى استخدام /start للتسجيل أولاً.")
        return
    
    purchases = fetch_query("""
        SELECT p.name, k.key_data, pur.purchase_date, pur.amount 
        FROM purchases pur
        JOIN products p ON pur.product_id = p.id
        JOIN keys k ON pur.key_id = k.id
        WHERE pur.user_id = ?
        ORDER BY pur.purchase_date DESC
    """, (message.chat.id,))
    
    if not purchases:
        bot.send_message(message.chat.id, "📭 **ليس لديك مشتريات حتى الآن.**", parse_mode="Markdown")
        return
    
    text = "📦 **مشترياتي السابقة:**\n\n"
    for i, (product, key, date, amount) in enumerate(purchases[:5], 1):
        text += f"{i}. 🎁 **{product}**\n"
        text += f"   🔑 المفتاح: `{key}`\n"
        text += f"   💰 السعر: ${amount:.2f}\n"
        text += f"   📅 التاريخ: {date}\n\n"
    
    if len(purchases) > 5:
        text += f"📌 وعرض {len(purchases)-5} مشتريات أخرى..."
    
    bot.send_message(message.chat.id, text, parse_mode="Markdown")

@bot.callback_query_handler(func=lambda call: True)
def handle_query(call):
    user = get_user(call.message.chat.id)
    if not user:
        bot.send_message(call.message.chat.id, "⚠️ يرجى استخدام /start للتسجيل أولاً.")
        return
    
    # قائمة الفئات
    if call.data.startswith('cat_'):
        category = call.data.replace('cat_', '')
        products = fetch_query("SELECT id, name, price, stock FROM products WHERE category=?", (category,))
        
        if not products:
            bot.answer_callback_query(call.id, "❌ لا يوجد منتجات في هذه الفئة")
            return
        
        markup = types.InlineKeyboardMarkup(row_width=1)
        for prod in products:
            markup.add(types.InlineKeyboardButton(
                f"{prod[1]} - ${prod[2]:.2f} ({prod[3]} متاح)", 
                callback_data=f"buy_{prod[0]}"
            ))
        markup.add(types.InlineKeyboardButton("🔙 رجوع", callback_data="main_menu"))
        
        bot.edit_message_text(f"🛍 **منتجات {category}:**", 
                             call.message.chat.id, call.message.message_id,
                             parse_mode="Markdown", reply_markup=markup)
    
    # شراء منتج
    elif call.data.startswith('buy_'):
        product_id = int(call.data.replace('buy_', ''))
        product = fetch_query("SELECT name, price FROM products WHERE id=?", (product_id,))
        
        if not product:
            bot.answer_callback_query(call.id, "❌ المنتج غير موجود")
            return
        
        # التحقق من وجود مفتاح متاح
        key = fetch_query("SELECT id, key_data FROM keys WHERE product_id=? AND status='available' LIMIT 1", 
                         (product_id,))
        
        if not key:
            bot.answer_callback_query(call.id, "❌ المنتج غير متوفر حالياً")
            return
        
        # عرض تأكيد الشراء
        markup = types.InlineKeyboardMarkup(row_width=2)
        markup.add(
            types.InlineKeyboardButton("✅ تأكيد الشراء", callback_data=f"confirm_{product_id}_{key[0][0]}"),
            types.InlineKeyboardButton("❌ إلغاء", callback_data="main_menu")
        )
        
        bot.edit_message_text(f"🛒 **تأكيد الشراء**\n\n"
                             f"المنتج: {product[0][0]}\n"
                             f"السعر: ${product[0][1]:.2f}\n"
                             f"رصيدك: ${user[2]:.2f}\n\n"
                             f"هل أنت متأكد من شراء هذا المنتج؟",
                             call.message.chat.id, call.message.message_id,
                             parse_mode="Markdown", reply_markup=markup)
    
    # تأكيد الشراء
    elif call.data.startswith('confirm_'):
        _, product_id, key_id = call.data.split('_')
        product_id = int(product_id)
        key_id = int(key_id)
        
        product = fetch_query("SELECT name, price FROM products WHERE id=?", (product_id,))
        if not product:
            bot.answer_callback_query(call.id, "❌ خطأ في المنتج")
            return
        
        price = product[0][1]
        
        if user[2] < price:
            bot.answer_callback_query(call.id, "❌ الرصيد غير كافٍ")
            return
        
        # تنفيذ الشراء
        execute_query("UPDATE users SET balance = balance - ? WHERE telegram_id = ?", 
                     (price, call.message.chat.id))
        execute_query("UPDATE keys SET status='sold', purchase_date=?, buyer_id=? WHERE id=?", 
                     (datetime.now().strftime("%Y-%m-%d %H:%M:%S"), call.message.chat.id, key_id))
        execute_query("INSERT INTO purchases (user_id, product_id, key_id, purchase_date, amount) VALUES (?, ?, ?, ?, ?)",
                     (call.message.chat.id, product_id, key_id, datetime.now().strftime("%Y-%m-%d %H:%M:%S"), price))

def process_add_product_name(message):
    name = message.text
    msg = bot.send_message(message.chat.id, "💰 أدخل سعر المنتج:")
    bot.register_next_step_handler(msg, lambda m: process_add_product_price(m, name))

def process_add_product_price(message, name):
    try:
        price = float(message.text)
        msg = bot.send_message(message.chat.id, "📂 أدخل فئة المنتج (ألعاب/برامج/تطبيقات/حسابات/أخرى):")
        bot.register_next_step_handler(msg, lambda m: process_add_product_category(m, name, price))
    except ValueError:

def process_add_product_category(message, name, price):
    category = message.text
    msg = bot.send_message(message.chat.id, "📦 أدخل كمية المنتج:")
    bot.register_next_step_handler(msg, lambda m: process_add_product_stock(m, name, price, category))

def process_add_product_stock(message, name, price, category):
    try:
        stock = int(message.text)
        execute_query("INSERT INTO products (name, price, category, stock) VALUES (?, ?, ?, ?)", 
                     (name, price, category, stock))
        bot.send_message(message.chat.id, f"✅ **تم إضافة المنتج بنجاح!**\n\n"
                         f"📝 الاسم: {name}\n"
                         f"💰 السعر: ${price:.2f}\n"
                         f"📂 الفئة: {category}\n"
                         f"📦 الكمية: {stock}", parse_mode="Markdown")
    except ValueError:
        bot.send_message(message.chat.id, "❌ يرجى إدخال رقم صحيح للكمية.")

def process_add_keys(message, product_id):
    keys = message.text.strip().split('\n')
    for key in keys:
        if key.strip():
            execute_query("INSERT INTO keys (product_id, key_data) VALUES (?, ?)", 
                         (product_id, key.strip()))
    
    # تحديث الكمية في المنتج
    execute_query("UPDATE products SET stock = stock + ? WHERE id = ?", (len(keys), product_id))
    
    bot.send_message(message.chat.id, f"✅ **تم إضافة {len(keys)} مفتاح/مفاتيح بنجاح!**", parse_mode="Markdown")

def process_add_balance(message, user_id):
    try:
        amount = float(message.text)
        user = get_user(user_id)
        if not user:
            bot.send_message(message.chat.id, "❌ المستخدم غير موجود")
            return
        
        execute_query("UPDATE users SET balance = balance + ? WHERE telegram_id = ?", (amount, user_id))
        bot.send_message(message.chat.id, f"✅ **تم شحن الرصيد بنجاح!**\n\n"
                         f"👤 المستخدم: {user[1]}\n"
                         f"💰 المبلغ: ${amount:.2f}\n"
                         f"📊 الرصيد الجديد: ${user[2] + amount:.2f}", parse_mode="Markdown")
        
        # إرسال إشعار للمستخدم
        try:
            bot.send_message(user_id, f"💰 **تم شحن رصيدك بمبلغ ${amount:.2f}**\n"
                             f"📊 الرصيد الحالي: ${user[2] + amount:.2f}", parse_mode="Markdown")
        except:
            pass
    except ValueError:
        bot.send_message(message.chat.id, "❌ يرجى إدخال رقم صحيح للمبلغ.")
