# ─── File Storage ───────────────────────────────────
def load_keys():
    if not os.path.exists(KEYS_FILE):
        return []
    with open(KEYS_FILE, "r", encoding="utf-8") as f:
        return json.load(f)

def save_keys(keys):
    with open(KEYS_FILE, "w", encoding="utf-8") as f:
        json.dump(keys, f, ensure_ascii=False, indent=2)

def generate_key():
    import random, string
    chars = string.ascii_uppercase + string.digits
    parts = ["".join(random.choices(chars, k=4)) for _ in range(4)]
    return "-".join(parts)

async def handle_key_input(update: Update, context: ContextTypes.DEFAULT_TYPE):
    if not context.user_data.get("waiting_for_key"):
        return

    key_input = update.message.text.strip().upper()
    context.user_data["waiting_for_key"] = False
    keys = load_keys()

    # Search for the key
    key_data = None
    for k in keys:
        if k["key"] == key_input:
            key_data = k
            break

    keyboard = InlineKeyboardMarkup([
        [InlineKeyboardButton("🔑 تفعيل مفتاح آخر", callback_data="redeem")],
        [InlineKeyboardButton("🏠 القائمة الرئيسية", callback_data="back_menu")],
    ])

    if not key_data:
        await update.message.reply_text(
            "❌ *مفتاح غير صالح*\n\n"
            "المفتاح الذي أدخلته غير موجود أو تم كتابته بشكل خاطئ.\n\n"
            "✅ تأكد من كتابة المفتاح بالشكل الصحيح.\n"
            "✅ المفاتيح تُصنع حصرياً بواسطة المطور @MESTRAXMODZ",
            parse_mode="Markdown",
            reply_markup=keyboard
        )
        return

    if key_data["used"]:
        used_by = key_data.get("used_by", "مستخدم سابقاً")
        used_at = key_data.get("used_at", "")
        await update.message.reply_text(
            "⚠️ *المفتاح مستخدم مسبقاً*\n\n"
            f"تم استخدام هذا المفتاح مسبقاً بواسطة: `{used_by}`\n"
            f"📅 تاريخ الاستخدام: {used_at}",
            parse_mode="Markdown",
            reply_markup=keyboard
        )
        return

    if key_data.get("expires_at"):
        exp = datetime.fromisoformat(key_data["expires_at"])
        if datetime.now() > exp:
            await update.message.reply_text(
                "⏰ *المفتاح منتهي الصلاحية*\n\n"
                f"تاريخ الانتهاء: {exp.strftime('%Y-%m-%d %H:%M')}",
                parse_mode="Markdown",
                reply_markup=keyboard
            )
            return

    # Activate
    user = update.effective_user
    key_data["used"] = True
    key_data["used_by"] = f"@{user.username}" if user.username else str(user.id)
    key_data["used_at"] = datetime.now().strftime("%Y-%m-%d %H:%M")
    save_keys(keys)

    duration = key_data.get("duration", "غير محدد")
    plan = key_data.get("plan", "عادي")
    expires = key_data.get("expires_at", "غير محدد")
    if expires != "غير محدد":
        expires = datetime.fromisoformat(expires).strftime("%Y-%m-%d %H:%M")

    await update.message.reply_text(
        "✅ *تم تفعيل المفتاح بنجاح!*\n\n"
        "━━━━━━━━━━━━━━━━━━━━\n"
        f"👤 المستخدم: `@{user.username}`" if user.username else f"👤 المستخدم: `{user.id}`" + "\n"
        f"🔑 المفتاح: `{key_input}`\n"
        f"📋 الخطة: {plan}\n"
        f"⏳ المدة: {duration}\n"
        f"📅 ينتهي في: {expires}\n"
        "━━━━━━━━━━━━━━━━━━━━\n\n"
        "🔒 هذا المفتاح مصنوع حصرياً بواسطة:\n"
        "👨‍💻 *@MESTRAXMODZ*",

@owner_only
async def gen_key(update: Update, context: ContextTypes.DEFAULT_TYPE):
    keys = load_keys()
    duration = context.args[0] if context.args else "30"

    if duration == "1":
        plan, days = "يوم واحد", 1
    elif duration == "7":
        plan, days = "أسبوع", 7
    elif duration == "30":
        plan, days = "شهر", 30
    else:
        try:
            days = int(duration)
            plan = f"{days} يوم"
        except ValueError:
            await update.message.reply_text("❌ استخدم: /gen 1 أو /gen 7 أو /gen 30")
            return

    new_key = generate_key()
    expires_at = (datetime.now() + timedelta(days=days)).isoformat()

    keys.append({
        "key": new_key,
        "plan": plan,
        "duration": f"{days} يوم",
        "expires_at": expires_at,
        "used": False,
        "used_by": None,
        "used_at": None,
        "created_at": datetime.now().strftime("%Y-%m-%d %H:%M")
    })
    save_keys(keys)

    await update.message.reply_text(
        f"✅ *تم إنشاء مفتاح جديد*\n\n"
        f"🔑 المفتاح: `{new_key}`\n"
        f"📋 الخطة: {plan}\n"
        f"📅 ينتهي في: {(datetime.now() + timedelta(days=days)).strftime('%Y-%m-%d %H:%M')}\n",
        parse_mode="Markdown"
    )

@owner_only
async def gen_bulk(update: Update, context: ContextTypes.DEFAULT_TYPE):
    keys = load_keys()
    count = 5
    duration = 30

    if context.args:
        try:
            count = int(context.args[0])
        except ValueError:
            pass
        if len(context.args) > 1:
            try:
                duration = int(context.args[1])
            except ValueError:
                pass

    count = min(count, 50)
    created = []

    for _ in range(count):
        new_key = generate_key()
        expires_at = (datetime.now() + timedelta(days=duration)).isoformat()
        keys.append({
            "key": new_key,
            "plan": f"{duration} يوم",
            "duration": f"{duration} يوم",
            "expires_at": expires_at,
            "used": False,
            "used_by": None,
            "used_at": None,
            "created_at": datetime.now().strftime("%Y-%m-%d %H:%M")
        })
        created.append(new_key)

    save_keys(keys)
    keys_text = "\n".join(f"  `{k}`" for k in created)

    await update.message.reply_text(
        f"✅ *تم إنشاء {count} مفاتيح* ({duration} يوم)\n\n"
        f"{keys_text}",
        parse_mode="Markdown"
    )

@owner_only
async def list_keys(update: Update, context: ContextTypes.DEFAULT_TYPE):
    keys = load_keys()
    if not keys:
        await update.message.reply_text("📋 لا توجد مفاتيح.")
        return

    active = [k for k in keys if not k["used"]]
    used = [k for k in keys if k["used"]]

    text = f"📊 *إحصائيات المفاتيح*\n\n"
    text += f"✅ نشطة: {len(active)}\n"
    text += f"❌ مستخدمة: {len(used)}\n"
    text += f"📦 الإجمالي: {len(keys)}\n\n"

    if active:
        text += "━━ *المفاتيح النشطة* ━━\n"
        for k in active:
            text += f"  `{k['key']}` | {k.get('plan', '')} | ينتهي: {k.get('expires_at', '')[:10]}\n"
        text += "\n"

    if used:
        text += "━━ *المفاتيح المستخدمة* ━━\n"
        for k in used:
            text += f"  `~~{k['key']}~~` | بواسطة: {k.get('used_by', '')}\n"

    await update.message.reply_text(text, parse_mode="Markdown")

@owner_only
async def delete_used(update: Update, context: ContextTypes.DEFAULT_TYPE):
    keys = load_keys()
    before = len(keys)
    keys = [k for k in keys if not k["used"]]
    deleted = before - len(keys)
    save_keys(keys)
    await update.message.reply_text(f"🗑️ تم حذف {deleted} مفتاح مستخدم.")

@owner_only
async def delete_all(update: Update, context: ContextTypes.DEFAULT_TYPE):
