def init_db():
    try:
        conn = sqlite3.connect(DB_PATH, check_same_thread=False)
        c = conn.cursor()
        # جدول الحسابات (المستخدمين المسموح لهم بالدخول)
        c.execute('''CREATE TABLE IF NOT EXISTS accounts 
                     (username TEXT PRIMARY KEY, password TEXT, telegram_id INTEGER, balance REAL DEFAULT 0.0, created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP)''')
        # الجداول الأساسية للمتجر
        c.execute('''CREATE TABLE IF NOT EXISTS categories (id INTEGER PRIMARY KEY AUTOINCREMENT, name TEXT)''')
        c.execute('''CREATE TABLE IF NOT EXISTS products (id INTEGER PRIMARY KEY AUTOINCREMENT, cat_id INTEGER, name TEXT, price REAL)''')
        c.execute('''CREATE TABLE IF NOT EXISTS stock (id INTEGER PRIMARY KEY AUTOINCREMENT, prod_id INTEGER, key_data TEXT, sold BOOLEAN DEFAULT 0, sold_to INTEGER)''')
        # جدول للمعاملات
        c.execute('''CREATE TABLE IF NOT EXISTS transactions 
                     (id INTEGER PRIMARY KEY AUTOINCREMENT, username TEXT, amount REAL, type TEXT, timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP)''')
        conn.commit()
        conn.close()
        print("✅ Database initialized successfully!")
        return True
    except Exception as e:
        print(f"❌ Database initialization error: {e}")
        return False

def execute_query(query, params=()):
    try:
        conn = sqlite3.connect(DB_PATH, check_same_thread=False)
        c = conn.cursor()
        c.execute(query, params)
        conn.commit()
        conn.close()
        return True
    except Exception as e:
        print(f"❌ Query execution error: {e}")
        return False

def fetch_query(query, params=()):
    try:
        conn = sqlite3.connect(DB_PATH, check_same_thread=False)
        c = conn.cursor()
        c.execute(query, params)
        result = c.fetchall()
        conn.close()
        return result
    except Exception as e:
        print(f"❌ Fetch query error: {e}")
        return []

# ================= دوال الأدمن (Next Step Handlers) =================
def process_create_login(message):
    try:
        username, password = message.text.split()
        if execute_query("INSERT INTO accounts (username, password) VALUES (?, ?)", (username, password)):
            bot.send_message(message.chat.id, f"✅ Account Created Successfully!\n- Username: `{username}`\n- Password: `{password}`", parse_mode="Markdown")
            # تسجيل العملية
            execute_query("INSERT INTO transactions (username, amount, type) VALUES (?, ?, ?)", (username, 0, 'create_account'))
        else:
            bot.send_message(message.chat.id, "❌ Error creating account.")
    except sqlite3.IntegrityError:
        bot.send_message(message.chat.id, "❌ Error: Username already exists.")
    except ValueError:
        bot.send_message(message.chat.id, "❌ Error: Format must be `Username Password`.", parse_mode="Markdown")
    except Exception as e:
        print(f"❌ Error in process_create_login: {e}")
        bot.send_message(message.chat.id, "❌ An error occurred. Please try again.")

def process_delete_login(message):
    try:
        username = message.text
        # التحقق من وجود الحساب
        account = fetch_query("SELECT username FROM accounts WHERE username=?", (username,))
        if account:
            if execute_query("DELETE FROM accounts WHERE username=?", (username,)):
                bot.send_message(message.chat.id, f"✅ Account `{username}` has been deleted.", parse_mode="Markdown")
                execute_query("INSERT INTO transactions (username, amount, type) VALUES (?, ?, ?)", (username, 0, 'delete_account'))
            else:
                bot.send_message(message.chat.id, "❌ Error deleting account.")
        else:
            bot.send_message(message.chat.id, "❌ Account not found.")
    except Exception as e:
        print(f"❌ Error in process_delete_login: {e}")
        bot.send_message(message.chat.id, "❌ An error occurred. Please try again.")

def process_add_balance(message):
    try:
        username, amount = message.text.split()
        amount = float(amount)
        current = fetch_query("SELECT balance, telegram_id FROM accounts WHERE username=?", (username,))
        if current:
            new_bal = current[0][0] + amount
            telegram_id = current[0][1]
            if execute_query("UPDATE accounts SET balance=? WHERE username=?", (new_bal, username)):
                execute_query("INSERT INTO transactions (username, amount, type) VALUES (?, ?, ?)", (username, amount, 'deposit'))
                bot.send_message(message.chat.id, f"✅ Added ${amount} to {username}. New Balance: ${new_bal}")
                if telegram_id:
                    bot.send_message(telegram_id, f"💰 Your account balance has been credited with **${amount}**.\nNew Balance: **${new_bal}**", parse_mode="Markdown")
            else:
                bot.send_message(message.chat.id, "❌ Error updating balance.")
        else:
            bot.send_message(message.chat.id, "❌ Account not found.")
    except ValueError:
        bot.send_message(message.chat.id, "❌ Error. Please enter a valid number.")
    except Exception as e:
        print(f"❌ Error in process_add_balance: {e}")
        bot.send_message(message.chat.id, "❌ Error. Format: `Username 15.5`", parse_mode="Markdown")
