async def callback_handler(update: Update, ctx: ContextTypes.DEFAULT_TYPE):
    query = update.callback_query
    data = query.data

    if data == "back_menu":
        await back_menu_cb(update, ctx)
    elif data == "activate_prompt":
        await activate_prompt_cb(update, ctx)
    elif data == "check_ip":
        await check_ip_cb(update, ctx)
    elif data == "server_status":
        await server_status_cb(update, ctx)
    elif data == "help":
        await help_cb(update, ctx)
    elif data == "admin_back":
        await admin_back_cb(update, ctx)
    elif data == "view_keys":
        await view_keys_cb(update, ctx)
    elif data == "delete_used":
        await delete_used_cb(update, ctx)
    elif data == "delete_all":
        await delete_all_cb(update, ctx)
    elif data.startswith("gen5_"):
        days = int(data.split("_")[1])
        await gen_key(update, ctx, count=5, days=days)
    elif data.startswith("gen_"):
        days = int(data.split("_")[1])
        await gen_key(update, ctx, count=1, days=days)
