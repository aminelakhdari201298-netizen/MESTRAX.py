async def on_button(update: Update, ctx: ContextTypes.DEFAULT_TYPE):
    query = update.callback_query
    await query.answer()
    uid = query.from_user.id
    data = query.data

    # ── Global navigation ──
    if data == "main_menu":
        await query.edit_message_text(MAIN_MENU, reply_markup=main_keyboard(), parse_mode="Markdown")
        ctx.user_data["state"] = "menu"

    elif data == "activate":
        await query.edit_message_text(ASK_KEY, reply_markup=back_keyboard(), parse_mode="Markdown")
        ctx.user_data["state"] = "waiting_key"

    elif data == "check_ip":
        await query.edit_message_text(
            CHECK_IP,
            reply_markup=InlineKeyboardMarkup([
                [InlineKeyboardButton("🔗 Show My IP", url="https://api.ipify.org?format=json")],
                [InlineKeyboardButton("🔙 Back", callback_data="main_menu")],
            ]),
            parse_mode="Markdown"
        )

    elif data == "stats":
        await query.edit_message_text(STATS_MSG, reply_markup=back_keyboard(), parse_mode="Markdown")

    elif data == "help":
        await query.edit_message_text(HELP_MSG, reply_markup=back_keyboard(), parse_mode="Markdown")

    elif data == "ask_pass":
        await query.edit_message_text(ASK_PASSWORD, reply_markup=back_keyboard(), parse_mode="Markdown")
        ctx.user_data["state"] = "waiting_pass"

    # ── Admin actions ──
    elif data == "gen_key":
        if uid != OWNER_ID:
            await query.answer("⛔ Not authorized!", show_alert=True)
            return
        # Ask for duration
        await query.edit_message_text(ASK_DURATION, reply_markup=duration_keyboard(), parse_mode="Markdown")
        ctx.user_data["state"] = "waiting_duration"

    # Duration selection
    elif data.startswith("dur_"):
        if uid != OWNER_ID:
            await query.answer("⛔ Not authorized!", show_alert=True)
            return
        days = int(data.split("_")[1])
        # Generate key
        key = gen_key()
        expiry = datetime.now() + timedelta(days=days)
        active_keys.append({"key": key, "expiry": expiry, "duration": f"{days} days"})
        clean_expired_keys()  # just in case
        await query.edit_message_text(
            f"🔑 *Key Generated!*\n\n`{key}`\n"
            f"⏳ Expires: {expiry.strftime('%Y-%m-%d %H:%M:%S')}\n"
            f"📅 Duration: {days} days\n\n"
            f"📊 Active: {len(active_keys)} | Used: {len(used_keys)}",
            reply_markup=gen_keyboard(),
            parse_mode="Markdown",
        )
        ctx.user_data["state"] = "owner"

    elif data == "view_keys":
        if uid != OWNER_ID:
            await query.answer("⛔ Not authorized!", show_alert=True)
            return
        clean_expired_keys()
        if not active_keys:
            msg = "📦 *No active keys.*\n\nGenerate new keys first."
        else:
            lines = []
            for idx, k in enumerate(active_keys, 1):
                exp = k["expiry"].strftime("%Y-%m-%d %H:%M")
                lines.append(f"{idx}. `{k['key']}` – expires {exp}")
            msg = f"📦 *Active Keys ({len(active_keys)})*\n\n" + "\n".join(lines)
        await query.edit_message_text(msg, reply_markup=admin_keyboard(), parse_mode="Markdown")

    elif data == "del_keys":
        if uid != OWNER_ID:
            await query.answer("⛔ Not authorized!", show_alert=True)
            return
        active_keys.clear()
        used_keys.clear()
        await query.edit_message_text(
            "🗑 *All keys deleted.*",
            reply_markup=admin_keyboard(),
            parse_mode="Markdown",
        )


async def on_message(update: Update, ctx: ContextTypes.DEFAULT_TYPE):
    state = ctx.user_data.get("state", "menu")
    uid = update.effective_user.id
    text = update.message.text.strip()

    # ── Password check ──
    if state == "waiting_pass":
        if text == OWNER_PASSWORD and uid == OWNER_ID:
            await update.message.reply_text(
                "✅ *Access Granted*\n\n🔐 Admin Panel:",
                reply_markup=admin_keyboard(),
                parse_mode="Markdown",
            )
            ctx.user_data["state"] = "owner"
        else:
            await update.message.reply_text(
                "❌ *Wrong password.*",
                reply_markup=main_keyboard(),
                parse_mode="Markdown",
            )
            ctx.user_data["state"] = "menu"
        return

    # ── Key activation flow ──
    if state == "waiting_key":
        # Check if key is valid and not expired
        clean_expired_keys()
        key_entry = next((k for k in active_keys if k["key"] == text), None)
        if not key_entry:
            # check if used
            used_entry = next((k for k in used_keys if k["key"] == text), None)
            if used_entry:
                await update.message.reply_text(USED_MSG, parse_mode="Markdown")
            else:
                reason = "Key not found or expired."
                await update.message.reply_text(
                    FAILED_MSG.format(reason=reason),
                    parse_mode="Markdown"
                )
            ctx.user_data["state"] = "menu"
            return

        # Key is valid, ask for IP
        ctx.user_data["pending_key"] = text
        await update.message.reply_text(
            ASK_IP,
            reply_markup=InlineKeyboardMarkup([
                [InlineKeyboardButton("🌐 Check My IP", callback_data="check_ip")],
                [InlineKeyboardButton("🔙 Cancel", callback_data="main_menu")],
            ]),
            parse_mode="Markdown"
        )
        ctx.user_data["state"] = "waiting_ip"
        return

    if state == "waiting_ip":
        # Validate IP format (simple)
        if not text.replace('.', '').replace(':', '').isdigit():  # crude check for IPv4/IPv6
            await update.message.reply_text(
                "❌ *Invalid IP format.* Please send a valid IP address.",
                parse_mode="Markdown"
            )
            return

        pending_key = ctx.user_data.get("pending_key")
        if not pending_key:
            await update.message.reply_text(
                "⚠️ *Session expired.* Please start over with /start.",
                parse_mode="Markdown"
            )
            ctx.user_data["state"] = "menu"
            return

        # Re-verify key (in case it expired or was used in between)
        clean_expired_keys()
        key_entry = next((k for k in active_keys if k["key"] == pending_key), None)
        if not key_entry:
            await update.message.reply_text(
                FAILED_MSG.format(reason="Key no longer valid."),
                parse_mode="Markdown"
            )
            ctx.user_data["state"] = "menu"
            return

        # Move key to used
        active_keys.remove(key_entry)
        used_keys.append({
            "key": pending_key,
            "user_id": uid,
            "ip": text,
            "activated_at": datetime.now(),
            "expiry": key_entry["expiry"]
        })

        # Show success with expiry info
        expires_str = key_entry["expiry"].strftime("%Y-%m-%d %H:%M:%S")
        await update.message.reply_text(
            SUCCESS_MSG.format(expires=expires_str),
            parse_mode="Markdown"
        )
        ctx.user_data["state"] = "menu"
        return

    # ── Default fallback ──
    await update.message.reply_text(MAIN_MENU, reply_markup=main_keyboard(), parse_mode="Markdown")
    ctx.user_data["state"] = "menu"


def main():
    print("🔥 MESTRAX Proxy Bot Starting...")
    print(f"   Owner: {OWNER_ID}")
    app = Application.builder().token(BOT_TOKEN).build()
    app.add_handler(CommandHandler("start", start))
    app.add_handler(CommandHandler("admin", admin_cmd))
    app.add_handler(CallbackQueryHandler(on_button))
    app.add_handler(MessageHandler(filters.TEXT & ~filters.COMMAND, on_message))
    print("✅ Running! Use /start or /admin")
    app.run_polling(allowed_updates=Update.ALL_TYPES)


if __name__ == "__main__":
